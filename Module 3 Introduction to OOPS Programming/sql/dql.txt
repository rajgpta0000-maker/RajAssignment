CREATE PROCEDURE productPriceIncre1()
BEGIN
    DECLARE p_id INT;
    DECLARE newPrice INT;
    DECLARE isDone INT DEFAULT 0;

    DECLARE productCur CURSOR FOR
        SELECT id, price FROM test;

    DECLARE CONTINUE HANDLER FOR NOT FOUND SET isDone = 1;

    OPEN productCur;

    readLoop: LOOP
        FETCH productCur INTO p_id, newPrice;

        IF isDone = 1 THEN
            LEAVE readLoop;
        END IF;

        UPDATE test
        SET price = newPrice + 200
        WHERE id = p_id;

    END LOOP;

    CLOSE productCur;
END //

DELIMITER ;

CALL productPriceIncre1();
